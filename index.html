Code sample

A simplified excerpt of the real Monetization module from Last Station. A purchase is never granted twice: the PurchaseId is stored in the player profile (ProfileStore). The PurchaseId is written to the profile before Roblox gets the confirmation. If the item can’t be granted or something fails, Roblox retries the receipt later.

--!strict
-- Monetization (simplified). The ONLY place where ProcessReceipt is set.
-- The PurchaseId goes into the profile (ProfileStore) so a purchase is never granted twice.
local MarketplaceService = game:GetService("MarketplaceService")
local Players = game:GetService("Players")

local CFG = GameConfig.Monetization

-- Grants a product. true = granted, false = can’t grant now (Roblox will retry later).
type ProductHandler = (player: Player) -> boolean

-- grantEmergencyBattery: returns false outside a match, otherwise adds station power
local productHandlers: { [number]: ProductHandler } = {
    [CFG.Products.EmergencyBattery.id] = grantEmergencyBattery,
}

local function rememberPurchase(purchaseIds: { string }, purchaseId: string)
    table.insert(purchaseIds, purchaseId)
    while #purchaseIds > CFG.maxStoredPurchaseIds do
        table.remove(purchaseIds, 1)
    end
end

local function processReceipt(receipt: { [string]: any }): Enum.ProductPurchaseDecision
    local player = Players:GetPlayerByUserId(receipt.PlayerId)
    if not player then
        return Enum.ProductPurchaseDecision.NotProcessedYet
    end

    local profile = MetaProfile.getProfile(player)
    if not profile or not profile:IsActive() then
        return Enum.ProductPurchaseDecision.NotProcessedYet
    end

    local purchaseIds: { string } = profile.Data.PurchaseIds
    local purchaseId: string = receipt.PurchaseId

    -- Already granted: just confirm
    if table.find(purchaseIds, purchaseId) then
        return Enum.ProductPurchaseDecision.PurchaseGranted
    end

    local handler = productHandlers[receipt.ProductId]
    if not handler then
        warn(`[Monetization] unknown product {receipt.ProductId}`)
        return Enum.ProductPurchaseDecision.NotProcessedYet
    end

    local ok, granted = pcall(handler, player)
    if not ok then
        warn(`[Monetization] grant failed {receipt.ProductId} for {player.Name}: {granted}`)
        return Enum.ProductPurchaseDecision.NotProcessedYet
    end
    if not granted then
        return Enum.ProductPurchaseDecision.NotProcessedYet
    end

    -- Remember the purchase in the profile first, then confirm to Roblox
    rememberPurchase(purchaseIds, purchaseId)
    return Enum.ProductPurchaseDecision.PurchaseGranted
end

MarketplaceService.ProcessReceipt = processReceipt
