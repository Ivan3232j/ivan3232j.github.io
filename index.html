Пример кода

Упрощённый пример: так устроены раунды в Last Station. Состояния меняются только по разрешённым переходам, клиенты узнают о смене через один RemoteEvent, а все цифры берутся из GameConfig.

--!strict
-- RoundManager: раунды как машина состояний
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local GameConfig = require(ReplicatedStorage.Shared.GameConfig)
local Remotes = require(ReplicatedStorage.Shared.Remotes)

export type State = "Waiting" | "Playing" | "Ended"

-- Из какого состояния в какое можно перейти. Остальные переходы запрещены
local TRANSITIONS: { [State]: { [State]: boolean } } = {
    Waiting = { Playing = true },
    Playing = { Ended = true },
    Ended = { Waiting = true },
}

local RoundManager = {}
local current: State = "Waiting"

function RoundManager.GetState(): State
    return current
end

function RoundManager.SetState(nextState: State)
    if not TRANSITIONS[current][nextState] then
        warn(`RoundManager: нельзя перейти {current} -> {nextState}`)
        return
    end
    print(`RoundManager: {current} -> {nextState}`)
    current = nextState
    Remotes.RoundState:FireAllClients(nextState)
end

-- Главный цикл: ждём игроков, играем раунд, показываем итоги
function RoundManager.Run()
    while true do
        repeat task.wait(1) until #Players:GetPlayers() >= GameConfig.MinPlayers

        RoundManager.SetState("Playing")
        task.wait(GameConfig.RoundLength)

        RoundManager.SetState("Ended")
        task.wait(GameConfig.IntermissionLength)

        RoundManager.SetState("Waiting")
    end
end

return RoundManager
