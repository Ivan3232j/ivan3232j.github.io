Last Station: Space Miners

Опубликована на Roblox. Кооп-выживание: игроки вылетают со станции, добывают ресурсы и не дают ей обесточиться. Вся техническая часть моя.

  • Матч на машине состояний RoundManager: экспедиция → тревога → оборона
  • Сервер решает всё: каждое действие клиента проверяется на сервере, стыковку со станцией тоже засчитывает сервер
  • Работает на телефонах: тач-управление и интерфейс, который подстраивается под экран
  • Баланс в одном модуле GameConfig, все RemoteEvents в одном модуле Remotes
  • Данные разделены на матч (RunState) и долгий прогресс (MetaProfile) на ProfileStore с блокировкой сессии
  • Монетизация: Game Pass и Developer Product через единый ProcessReceipt, одна покупка не выдаётся дважды
  • Аналитика через AnalyticsService: воронка новичка и экономика
  • Опасности: астероиды, радиация, солнечные вспышки; устаревший BodyGyro заменил на AlignOrientation

Typed Luau, ProfileStore, MarketplaceService, AnalyticsService, Rojo, Git

Doodle Dream Flight ✏️

Опубликована на Roblox. Полёт во сне по миру детских рисунков: пролетаешь кольца, проскакиваешь в пасти монстров, делаешь трюки и собираешь комбо.

  • Трассы строит генератор по seed, валидатор проверяет проходимость: тест прогоняет 60 трасс, сложность растёт от этапа к этапу

  • Пасти монстров открываются по общему серверному времени, сервер проверяет пролёт с допуском на пинг
  • Сервер считает очки и комбо, проверяет скорость (античит) и не даёт фармить трюки

  • Английский и русский: все тексты в одном модуле, шрифты подбираются под язык
  • Аналитика через AnalyticsService: воронка новичка, прогрессия этапов, экономика
  • Моя роль в работе с AI: писал ТЗ, ставил задачи AI-агентам (Gemini, Claude Code), ревьюил их код и тестировал игру в Studio через MCP

Пример кода

Упрощённый кусок настоящего модуля Monetization из Last Station. Покупка не выдаётся дважды: PurchaseId хранится в профиле игрока (ProfileStore). PurchaseId записывается в профиль раньше, чем Roblox получает подтверждение. Если выдать нельзя или что-то сломалось, Roblox повторит чек позже.

--!strict
-- Monetization (упрощённо). ЕДИНСТВЕННОЕ место, где задаётся ProcessReceipt.
-- PurchaseId пишется в профиль (ProfileStore), чтобы одна покупка не выдалась дважды.
local MarketplaceService = game:GetService("MarketplaceService")
local Players = game:GetService("Players")

local CFG = GameConfig.Monetization

-- Выдача товара. true — выдано, false — выдать нельзя сейчас (Roblox повторит позже).
type ProductHandler = (player: Player) -> boolean

-- grantEmergencyBattery: вне матча возвращает false, иначе добавляет станции энергию
local productHandlers: { [number]: ProductHandler } = {
    [CFG.Products.EmergencyBattery.id] = grantEmergencyBattery,
}

local function rememberPurchase(purchaseIds: { string }, purchaseId: string)
    table.insert(purchaseIds, purchaseId)
    while #purchaseIds > CFG.maxStoredPurchaseIds do
        table.remove(purchaseIds, 1)
    end
end

local function processReceipt(receipt: { [string]: any }): Enum.ProductPurchaseDecision
    local player = Players:GetPlayerByUserId(receipt.PlayerId)
    if not player then
        return Enum.ProductPurchaseDecision.NotProcessedYet
    end

    local profile = MetaProfile.getProfile(player)
    if not profile or not profile:IsActive() then
        return Enum.ProductPurchaseDecision.NotProcessedYet
    end

    local purchaseIds: { string } = profile.Data.PurchaseIds
    local purchaseId: string = receipt.PurchaseId

    -- Уже выдавали — просто подтверждаем
    if table.find(purchaseIds, purchaseId) then
        return Enum.ProductPurchaseDecision.PurchaseGranted
    end

    local handler = productHandlers[receipt.ProductId]
    if not handler then
        warn(`[Monetization] неизвестный товар {receipt.ProductId}`)
        return Enum.ProductPurchaseDecision.NotProcessedYet
    end

    local ok, granted = pcall(handler, player)
    if not ok then
        warn(`[Monetization] ошибка выдачи {receipt.ProductId} для {player.Name}: {granted}`)
        return Enum.ProductPurchaseDecision.NotProcessedYet
    end
    if not granted then
        return Enum.ProductPurchaseDecision.NotProcessedYet
    end

    -- Сначала запоминаем покупку в профиле, потом подтверждаем Roblox
    rememberPurchase(purchaseIds, purchaseId)
    return Enum.ProductPurchaseDecision.PurchaseGranted
end

MarketplaceService.ProcessReceipt = processReceipt
